# gRPC Foundation and Transports Implementation Plan (Plan 1 of 6)

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Give the app a working, env-selected gRPC and gRPC-Web transport layer (codegen, native Android gRPC module, gRPC-Web over `expo/fetch`, auth, error mapping, dev TLS trust), proven by an anonymous smoke call, with `Rest` mode unchanged.

**Architecture:** `protobuf-ts` generates TypeScript messages and clients from vendored backend protos. Generated clients run on an `RpcTransport`: `NativeGrpcTransport` (a byte-passthrough bridge to a local Expo Kotlin module on `grpc-okhttp`) in `Grpc` mode, or `GrpcWebFetchTransport` on `expo/fetch` in `GrpcWeb` mode. `GrpcClientProvider` owns the transport and exposes `call()`, which wraps every unary call with refresh-and-retry, consent logout and `ApiError` decoding.

**Tech Stack:** Expo SDK 54, React Native 0.81.5, TypeScript 5.9, `@protobuf-ts/*` 2.11.1, `grpc-okhttp` 1.73.0, Kotlin Expo Modules API, Jest via `jest-expo` ~54.

**Spec:** `docs/superpowers/specs/2026-10-11-grpc-transports-design.md`

**Plan series:** This plan covers spec delivery steps 1 (Foundation) and 2 (Transports). Later plans, written after this one lands: Plan 2 auth/users/legal/consents (step 3), Plan 3 live updates (step 4), Plan 4 homes (step 5), Plan 5 devices (step 6), Plan 6 automations, data sources and finish (steps 7–8). The `ApiProvider` skeleton moves to Plan 2, where its first feature API lands; it would be empty here.

## Global Constraints

- `EXPO_PUBLIC_TRANSPORT` = `Rest` | `Grpc` | `GrpcWeb`; unset means `Rest`; invalid logs an error and falls back to `Rest`. Read once, env-only, no runtime switch.
- All three modes are supported long-term. No REST or SignalR code is deleted or changed in behavior. `Rest` mode must work identically after every task.
- Android only. No iOS code in `modules/senswave-grpc`.
- Backend `.proto` files are the single source of truth; no hand-written protobuf code. `protos/` and `src/generated/` are committed.
- Read `process.env.EXPO_PUBLIC_*` only by direct property access (Expo inlines them at build time).
- Default unary deadline 15 000 ms. No default deadline on streams.
- Native gRPC requires an `https://` API URL.
- Error trailers: `senswave-errors-bin` (one entry per error, each a serialized `senswave.common.v1.Error`), `senswave-trace-id`.
- Commit and PR titles: `feat: …` / `fix: …` / `chore: …`. **No Claude attribution lines in commits or PRs.**
- Follow repo conventions: 4-space indent, double quotes, Prettier via `npm run fix`, `FC<Props>` components, `@/` alias for `src/`.

```ts
export type TransportMode = "Rest" | "Grpc" | "GrpcWeb";

const transportModes: readonly TransportMode[] = ["Rest", "Grpc", "GrpcWeb"];

export const parseTransportMode = (value: string | undefined): TransportMode => {
    if (!value) {
        return "Rest";
    }

    const mode = transportModes.find((m) => m === value);

    if (!mode) {
        console.error(
            `[Transport] Invalid EXPO_PUBLIC_TRANSPORT "${value}". Expected Rest, Grpc or GrpcWeb. Falling back to Rest.`,
        );
        return "Rest";
    }

    return mode;
};

// Direct property access: Expo inlines EXPO_PUBLIC_* variables at build time.
export const transportMode: TransportMode = parseTransportMode(process.env.EXPO_PUBLIC_TRANSPORT);

export const isGrpcMode = (mode: TransportMode): boolean => mode !== "Rest";

export const validateBaseUrlForMode = (mode: TransportMode, baseUrl: URL): string | null => {
    if (isGrpcMode(mode) && baseUrl.protocol !== "https:") {
        return `Transport ${mode} requires an https:// API URL, got ${baseUrl.origin}.`;
    }

    return null;
};
```

```js
// Regenerates src/generated/ from protos/ with the protobuf-ts plugin.
const { execFileSync } = require("child_process");
const fs = require("fs");
const path = require("path");

const root = path.resolve(__dirname, "..");
const protoDir = path.join(root, "protos");
const outDir = path.join(root, "src", "generated");

const listProtos = (dir) =>
    fs.readdirSync(dir, { withFileTypes: true }).flatMap((entry) => {
        const full = path.join(dir, entry.name);
        if (entry.isDirectory()) return listProtos(full);
        return entry.name.endsWith(".proto") ? [full] : [];
    });

const files = listProtos(protoDir)
    .map((file) => path.relative(root, file).split(path.sep).join("/"))
    .sort();

if (files.length === 0) {
    console.error(`No .proto files found in ${protoDir}. Run npm run proto:sync first.`);
    process.exit(1);
}

fs.rmSync(outDir, { recursive: true, force: true });
fs.mkdirSync(outDir, { recursive: true });

execFileSync(
    "npx",
    [
        "protoc",
        "--ts_out",
        "src/generated",
        "--ts_opt",
        "long_type_number,generate_dependencies",
        "--proto_path",
        "protos",
        ...files,
    ],
    { cwd: root, stdio: "inherit", shell: process.platform === "win32" },
);

console.log(`Generated ${files.length} proto files into src/generated.`);
```

```ts
export interface ApiErrorDetail {
    code: string;
    description: string;
}

/** Transport-neutral error: built from a REST ProblemDetails body or from gRPC status and trailers. */
export interface ApiError {
    statusText: string;
    errors: ApiErrorDetail[];
    traceId?: string;
}

export const connectionApiError: ApiError = {
    statusText: "Connection error. Please check your internet connection.",
    errors: [],
};

export const apiErrorMessage = (error: ApiError): string => {
    const first = error.errors[0];

    if (first) {
        return first.description ? first.description : first.code;
    }

    return "An unexpected error occurred.";
};
```

- [ ] **Step 4: Implement `src/api/rest/problemDetails.ts`**

```ts
import { ApiError } from "@/api/apiError";

interface ProblemDetailsBody {
    title?: string;
    errors?: unknown;
    traceId?: string;
}

interface ProblemDetailsError {
    code?: string;
    description?: string;
}

export const apiErrorFromProblemDetails = (text: string): ApiError | null => {
    let body: ProblemDetailsBody;

    try {
        body = JSON.parse(text);
    } catch {
        return null;
    }

    if (!body || typeof body !== "object") {
        return null;
    }

    const errors = Array.isArray(body.errors) ? (body.errors as ProblemDetailsError[]) : [];

    return {
        statusText: body.title ?? "",
        errors: errors.map((e) => ({ code: e.code ?? "", description: e.description ?? "" })),
        traceId: body.traceId,
    };
};
```

```ts
        if (httpResponse.statusCode === 401) return;

        if (httpResponse.response === null) {
            error("Connection error. Please check your internet connection.");
            return;
        }

        let text: string | undefined;
        try {

            error("An unexpected error occurred.");
            return;
        }

        apiError(parsed);
    };
```

```ts
import { base64decode } from "@protobuf-ts/runtime";
import { RpcError, RpcMetadata } from "@protobuf-ts/runtime-rpc";
import { Error as ProtoError } from "@/generated/common/v1/common";
import { ApiError, ApiErrorDetail } from "@/api/apiError";

export const errorsTrailerKey = "senswave-errors-bin";
export const traceIdTrailerKey = "senswave-trace-id";

// A repeated trailer arrives as an array (native) or as one comma-joined string (gRPC-Web).
// Base64 never contains commas, so splitting is safe.
const metadataValues = (meta: RpcMetadata, key: string): string[] => {
    const value = meta[key];

    if (value === undefined) {
        return [];
    }

    const list = Array.isArray(value) ? value : [value];

    return list
        .flatMap((v) => v.split(","))
        .map((v) => v.trim())
        .filter((v) => v !== "");
};

export const decodeErrorTrailers = (meta: RpcMetadata): ApiErrorDetail[] =>
    metadataValues(meta, errorsTrailerKey).flatMap((value) => {
        try {
            const error = ProtoError.fromBinary(base64decode(value));
            return [{ code: error.code, description: error.description }];
        } catch (e) {
            console.warn(`[gRPC] Failed to decode ${errorsTrailerKey} entry`, e);
            return [];
        }
    });

export const apiErrorFromRpcError = (error: RpcError): ApiError => ({
    statusText: error.message,
    errors: decodeErrorTrailers(error.meta ?? {}),
    traceId: metadataValues(error.meta ?? {}, traceIdTrailerKey)[0],
});
```

```ts
export type GrpcFailureKind =
    | "refreshAndRetry"
    | "emailNotConfirmed"
    | "invalidCredentials"
    | "consentMissing"
    | "homeAccessDenied"
    | "requestFailed"
    | "connectionError"
    | "timeout";

export interface GrpcFailureInput {
    code: string;
    serviceTypeName: string;
    methodName: string;
    detail: string;
}

const authService = "senswave.users.v1.AuthService";
const legalService = "senswave.users.v1.LegalService";
const liveUpdatesService = "senswave.liveupdates.v1.LiveUpdatesService";

// Packages LegalConsentInterceptor never checks (users, auth, legal, live updates).
const consentExemptPackages = ["senswave.users.v1.", "senswave.liveupdates.v1."];

export const isAnonymousService = (serviceTypeName: string): boolean =>
    serviceTypeName === authService || serviceTypeName === legalService;

/** Maps a failed gRPC call to the app behavior its REST equivalent triggers today. */
export const classifyGrpcFailure = ({
    code,
    serviceTypeName,
    methodName,
    detail,
}: GrpcFailureInput): GrpcFailureKind => {
    switch (code) {
        case "UNAUTHENTICATED":
            if (serviceTypeName !== authService) {
                return "refreshAndRetry";
            }
            if (methodName === "Login") {
                return detail === "NotAllowed" ? "emailNotConfirmed" : "invalidCredentials";
            }
            return "requestFailed";
        case "PERMISSION_DENIED":
            if (serviceTypeName === liveUpdatesService) {
                return "homeAccessDenied";
            }
            if (consentExemptPackages.some((p) => serviceTypeName.startsWith(p))) {
                return "requestFailed";
            }
            return "consentMissing";
        case "UNAVAILABLE":
            return "connectionError";
        case "DEADLINE_EXCEEDED":
            return "timeout";
        default:
            return "requestFailed";
    }
};
```

```js
// Trusts certs/dev-ca.crt for the hosts in EXPO_PUBLIC_DEV_CA_HOSTS, in every build type.
// Production hosts are never listed, so they keep system CAs only.
const { AndroidConfig, withAndroidManifest, withDangerousMod } = require("expo/config-plugins");
const fs = require("fs");
const path = require("path");

const certSource = path.join("certs", "dev-ca.crt");
const rawName = "senswave_dev_ca";
const xmlName = "network_security_config";

const parseHosts = (value) =>
    (value ?? "")
        .split(",")
        .map((host) => host.trim())
        .filter((host) => host !== "");

const buildNetworkSecurityConfig = ({ hosts, cleartextPermitted }) =>
    [
        '<?xml version="1.0" encoding="utf-8"?>',
        "<network-security-config>",
        `    <base-config cleartextTrafficPermitted="${cleartextPermitted}">`,
        "        <trust-anchors>",
        '            <certificates src="system" />',
        "        </trust-anchors>",
        "    </base-config>",
        "    <domain-config>",
        ...hosts.map((host) => `        <domain includeSubdomains="false">${host}</domain>`),
        "        <trust-anchors>",
        `            <certificates src="@raw/${rawName}" />`,
        '            <certificates src="system" />',
        "        </trust-anchors>",
        "    </domain-config>",
        "</network-security-config>",
        "",
    ].join("\n");

const writeFile = (file, content) => {
    fs.mkdirSync(path.dirname(file), { recursive: true });
    fs.writeFileSync(file, content);
};

const withDevCaTrust = (config) => {
    const hosts = parseHosts(process.env.EXPO_PUBLIC_DEV_CA_HOSTS);
    const certPath = path.resolve(process.cwd(), certSource);

    if (hosts.length === 0 || !fs.existsSync(certPath)) {
        console.warn(
            `[withDevCaTrust] Skipped: needs EXPO_PUBLIC_DEV_CA_HOSTS and ${certSource}. System CAs only.`,
        );
        return config;
    }

    config = withDangerousMod(config, [
        "android",
        async (cfg) => {
            const src = path.join(cfg.modRequest.platformProjectRoot, "app", "src");

            fs.mkdirSync(path.join(src, "main", "res", "raw"), { recursive: true });
            fs.copyFileSync(certPath, path.join(src, "main", "res", "raw", `${rawName}.crt`));

            // Release keeps cleartext off; debug keeps it on for Metro, as Expo's debug manifest does.
            writeFile(
                path.join(src, "main", "res", "xml", `${xmlName}.xml`),
                buildNetworkSecurityConfig({ hosts, cleartextPermitted: false }),
            );
            writeFile(
                path.join(src, "debug", "res", "xml", `${xmlName}.xml`),
                buildNetworkSecurityConfig({ hosts, cleartextPermitted: true }),
            );

            return cfg;
        },
    ]);

    return withAndroidManifest(config, (cfg) => {
        const application = AndroidConfig.Manifest.getMainApplicationOrThrow(cfg.modResults);
        application.$["android:networkSecurityConfig"] = `@xml/${xmlName}`;
        return cfg;
    });
};

module.exports = withDevCaTrust;
module.exports.parseHosts = parseHosts;
module.exports.buildNetworkSecurityConfig = buildNetworkSecurityConfig;
```

```kotlin
package expo.modules.senswavegrpc

import android.util.Base64
import io.grpc.Metadata
import io.grpc.MethodDescriptor
import java.io.ByteArrayInputStream
import java.io.InputStream

/** Passes serialized protobuf bytes through untouched; JS owns (de)serialization. */
internal object ByteArrayMarshaller : MethodDescriptor.Marshaller<ByteArray> {
    override fun stream(value: ByteArray): InputStream = ByteArrayInputStream(value)
    override fun parse(stream: InputStream): ByteArray = stream.readBytes()
}

/** Converts between grpc-java Metadata and the bridge shape: lowercase keys, value lists, base64 for "-bin". */
internal object MetadataCodec {
    fun encode(bytes: ByteArray): String = Base64.encodeToString(bytes, Base64.NO_WRAP)

    fun decode(value: String): ByteArray = Base64.decode(value, Base64.NO_WRAP)

    fun fromMap(map: Map<String, List<String>>): Metadata {
        val metadata = Metadata()
        for ((rawKey, values) in map) {
            val key = rawKey.lowercase()
            if (key.endsWith(Metadata.BINARY_HEADER_SUFFIX)) {
                val binaryKey = Metadata.Key.of(key, Metadata.BINARY_BYTE_MARSHALLER)
                values.forEach { metadata.put(binaryKey, decode(it)) }
            } else {
                val asciiKey = Metadata.Key.of(key, Metadata.ASCII_STRING_MARSHALLER)
                values.forEach { metadata.put(asciiKey, it) }
            }
        }
        return metadata
    }

    fun toMap(metadata: Metadata): Map<String, List<String>> = metadata.keys().associateWith { key ->
        if (key.endsWith(Metadata.BINARY_HEADER_SUFFIX)) {
            metadata.getAll(Metadata.Key.of(key, Metadata.BINARY_BYTE_MARSHALLER))?.map(::encode) ?: emptyList()
        } else {
            metadata.getAll(Metadata.Key.of(key, Metadata.ASCII_STRING_MARSHALLER))?.toList() ?: emptyList()
        }
    }
}
```

```kotlin
import expo.modules.kotlin.modules.ModuleDefinition

import io.grpc.MethodDescriptor

import java.net.URI
import java.util.concurrent.ConcurrentHashMap
import java.util.concurrent.TimeUnit

/**
 * Generic native gRPC client: JS sends serialized request bytes plus metadata for a full method
 * name, this module runs the call on a shared OkHttp channel and returns bytes, status and trailers.
 */
class SenswaveGrpcModule : Module() {
    private val channels = ConcurrentHashMap<String, ManagedChannel>()
    private val descriptors = ConcurrentHashMap<String, MethodDescriptor<ByteArray, ByteArray>>()
    private val calls = ConcurrentHashMap<String, ClientCall<ByteArray, ByteArray>>()

    override fun definition() = ModuleDefinition {
        Name("SenswaveGrpc")

        Events("onStreamHeaders", "onStreamMessage", "onStreamClose")

        AsyncFunction("unary") { callId: String, baseUrl: String, method: String, requestBase64: String,
                                 metadata: Map<String, List<String>>, deadlineMs: Double, promise: Promise ->
            var headers: Map<String, List<String>> = emptyMap()
            var response: ByteArray? = null

            val listener = object : ClientCall.Listener<ByteArray>() {
                override fun onHeaders(received: Metadata) {
                    headers = MetadataCodec.toMap(received)
                }

                override fun onMessage(message: ByteArray) {
                    response = message
                }

                override fun onClose(status: Status, trailers: Metadata) {
                    calls.remove(callId)
                    promise.resolve(
                        closeResult(status, trailers, headers) +
                            ("responseBase64" to response?.let(MetadataCodec::encode))
                    )
                }
            }

            startCall(callId, baseUrl, method, MethodDescriptor.MethodType.UNARY, requestBase64, metadata, deadlineMs, listener)
        }

        Function("startServerStream") { callId: String, baseUrl: String, method: String, requestBase64: String,
                                        metadata: Map<String, List<String>>, deadlineMs: Double ->
            var headers: Map<String, List<String>> = emptyMap()

            val listener = object : ClientCall.Listener<ByteArray>() {
                override fun onHeaders(received: Metadata) {
                    headers = MetadataCodec.toMap(received)
                    sendEvent("onStreamHeaders", mapOf("callId" to callId, "headers" to headers))
                }

                override fun onMessage(message: ByteArray) {
                    sendEvent("onStreamMessage", mapOf("callId" to callId, "base64" to MetadataCodec.encode(message)))
                    calls[callId]?.request(1)
                }

                override fun onClose(status: Status, trailers: Metadata) {
                    calls.remove(callId)
                    sendEvent("onStreamClose", closeResult(status, trailers, headers) + ("callId" to callId))
                }
            }

            startCall(callId, baseUrl, method, MethodDescriptor.MethodType.SERVER_STREAMING, requestBase64, metadata, deadlineMs, listener)
        }

        Function("cancel") { callId: String ->
            calls.remove(callId)?.cancel("Cancelled by client", null)
        }

        OnDestroy {
            calls.values.forEach { it.cancel("Module destroyed", null) }
            calls.clear()
            channels.values.forEach { it.shutdownNow() }
            channels.clear()
        }
    }

    private fun closeResult(status: Status, trailers: Metadata, headers: Map<String, List<String>>): Map<String, Any?> =
        mapOf(
            "status" to status.code.value(),
            "message" to (status.description ?: ""),
            "headers" to headers,
            "trailers" to MetadataCodec.toMap(trailers),
        )

    private fun startCall(
        callId: String,
        baseUrl: String,
        method: String,
        type: MethodDescriptor.MethodType,
        requestBase64: String,
        metadata: Map<String, List<String>>,
        deadlineMs: Double,
        listener: ClientCall.Listener<ByteArray>,
    ) {
        val call: ClientCall<ByteArray, ByteArray>
        val requestMetadata: Metadata
        val request: ByteArray

        try {
            var options = CallOptions.DEFAULT
            if (deadlineMs > 0) {
                options = options.withDeadlineAfter(deadlineMs.toLong(), TimeUnit.MILLISECONDS)
            }
            requestMetadata = MetadataCodec.fromMap(metadata)
            request = MetadataCodec.decode(requestBase64)
            call = channelFor(baseUrl).newCall(descriptorFor(method, type), options)
        } catch (e: Exception) {
            listener.onClose(Status.UNAVAILABLE.withDescription(e.message ?: "Failed to start call").withCause(e), Metadata())
            return
        }

        calls[callId] = call
        call.start(listener, requestMetadata)
        call.request(if (type == MethodDescriptor.MethodType.UNARY) 2 else 1)
        call.sendMessage(request)
        call.halfClose()
    }

    private fun descriptorFor(method: String, type: MethodDescriptor.MethodType): MethodDescriptor<ByteArray, ByteArray> =
        descriptors.getOrPut("$type:$method") {
            MethodDescriptor.newBuilder(ByteArrayMarshaller, ByteArrayMarshaller)
                .setType(type)
                .setFullMethodName(method)
                .build()
        }

    private fun channelFor(baseUrl: String): ManagedChannel {
        val uri = URI(baseUrl)
        val host = uri.host ?: throw IllegalArgumentException("Invalid base URL: $baseUrl")
        val port = if (uri.port != -1) uri.port else 443

        return channels.getOrPut("$host:$port") {
            // TLS through the platform trust manager, so network_security_config applies.
            OkHttpChannelBuilder.forAddress(host, port).useTransportSecurity().build()
        }
    }
}
```

```ts
import { base64decode, base64encode } from "@protobuf-ts/runtime";
import {
    ClientStreamingCall,
    Deferred,
    DuplexStreamingCall,
    mergeRpcOptions,
    MethodInfo,
    RpcError,
    RpcInterceptor,
    RpcMetadata,
    RpcOptions,
    RpcOutputStreamController,
    RpcStatus,
    RpcTransport,
    ServerStreamingCall,
    UnaryCall,
} from "@protobuf-ts/runtime-rpc";
import { GrpcStatusCode } from "@protobuf-ts/grpcweb-transport";
import { NativeCallResult, NativeGrpcBridge, NativeMetadata, NativeSubscription } from "./nativeGrpcBridge";

export interface NativeGrpcTransportOptions {
    bridge: NativeGrpcBridge;
    /** API origin, for example "https://localhost:7070". Read per call so URL overrides apply. */
    getBaseUrl: () => string;
    interceptors?: RpcInterceptor[];
}

let callCounter = 0;
const nextCallId = () => `grpc-${Date.now()}-${++callCounter}`;

const toNativeMetadata = (meta: RpcMetadata | undefined): NativeMetadata =>
    Object.fromEntries(
        Object.entries(meta ?? {}).map(([key, value]) => [
            key.toLowerCase(),
            Array.isArray(value) ? value : [value],
        ]),
    );

const fromNativeMetadata = (meta: NativeMetadata): RpcMetadata =>
    Object.fromEntries(
        Object.entries(meta).map(([key, values]) => [key, values.length === 1 ? values[0] : values]),
    );

const deadlineMs = (timeout: RpcOptions["timeout"]): number => {
    if (timeout === undefined) return 0;
    if (typeof timeout === "number") return timeout;
    return Math.max(1, timeout.getTime() - Date.now());
};

const statusName = (code: number): string => GrpcStatusCode[code] ?? "UNKNOWN";

const methodPath = (method: MethodInfo) => `${method.service.typeName}/${method.name}`;

const rpcError = (method: MethodInfo, result: NativeCallResult): RpcError => {
    const error = new RpcError(result.message, statusName(result.status), fromNativeMetadata(result.trailers));
    error.serviceName = method.service.typeName;
    error.methodName = method.name;
    return error;
};

/** protobuf-ts RpcTransport over modules/senswave-grpc (native gRPC, HTTP/2, Android). */
export class NativeGrpcTransport implements RpcTransport {
    private readonly defaultOptions: RpcOptions;

    constructor(private readonly options: NativeGrpcTransportOptions) {
        this.defaultOptions = { interceptors: options.interceptors };
    }

    mergeOptions(options?: Partial<RpcOptions>): RpcOptions {
        return mergeRpcOptions(this.defaultOptions, options);
    }

    unary<I extends object, O extends object>(
        method: MethodInfo<I, O>,
        input: I,
        options: RpcOptions,
    ): UnaryCall<I, O> {
        const callId = nextCallId();
        const requestMeta = options.meta ?? {};
        const headers = new Deferred<RpcMetadata>();
        const response = new Deferred<O>();
        const status = new Deferred<RpcStatus>();
        const trailers = new Deferred<RpcMetadata>();

        const fail = (error: RpcError) => {
            headers.rejectPending(error);
            response.rejectPending(error);
            status.rejectPending(error);
            trailers.rejectPending(error);
        };

        options.abort?.addEventListener("abort", () => this.options.bridge.cancel(callId));

        this.options.bridge
            .unary(
                callId,
                this.options.getBaseUrl(),
                methodPath(method),
                base64encode(method.I.toBinary(input, options.binaryOptions)),
                toNativeMetadata(requestMeta),
                deadlineMs(options.timeout),
            )
            .then(
                (result) => {
                    headers.resolvePending(fromNativeMetadata(result.headers));

                    if (result.status !== GrpcStatusCode.OK || result.responseBase64 === null) {
                        fail(rpcError(method, result));
                        return;
                    }

                    response.resolvePending(
                        method.O.fromBinary(base64decode(result.responseBase64), options.binaryOptions),
                    );
                    status.resolvePending({ code: "OK", detail: result.message });
                    trailers.resolvePending(fromNativeMetadata(result.trailers));
                },
                (reason) => {
                    const error = new RpcError(String(reason), "UNAVAILABLE");
                    error.serviceName = method.service.typeName;
                    error.methodName = method.name;
                    fail(error);
                },
            );

        return new UnaryCall<I, O>(
            method,
            requestMeta,
            input,
            headers.promise,
            response.promise,
            status.promise,
            trailers.promise,
        );
    }

    serverStreaming<I extends object, O extends object>(
        method: MethodInfo<I, O>,
        input: I,
        options: RpcOptions,
    ): ServerStreamingCall<I, O> {
        const callId = nextCallId();
        const requestMeta = options.meta ?? {};
        const bridge = this.options.bridge;
        const headers = new Deferred<RpcMetadata>();
        const responses = new RpcOutputStreamController<O>();
        const status = new Deferred<RpcStatus>();
        const trailers = new Deferred<RpcMetadata>();

        const subscriptions: NativeSubscription[] = [
            bridge.addListener("onStreamHeaders", (event) => {
                if (event.callId === callId) headers.resolvePending(fromNativeMetadata(event.headers));
            }),
            bridge.addListener("onStreamMessage", (event) => {
                if (event.callId !== callId) return;
                responses.notifyMessage(method.O.fromBinary(base64decode(event.base64), options.binaryOptions));
            }),
            bridge.addListener("onStreamClose", (event) => {
                if (event.callId !== callId) return;
                subscriptions.forEach((s) => s.remove());
                headers.resolvePending(fromNativeMetadata(event.headers));

                if (event.status === GrpcStatusCode.OK) {
                    responses.notifyComplete();
                    status.resolvePending({ code: "OK", detail: event.message });
                    trailers.resolvePending(fromNativeMetadata(event.trailers));
                    return;
                }

                const error = rpcError(method, event);
                responses.notifyError(error);
                status.rejectPending(error);
                trailers.rejectPending(error);
            }),
        ];

        options.abort?.addEventListener("abort", () => bridge.cancel(callId));

        bridge.startServerStream(
            callId,
            this.options.getBaseUrl(),
            methodPath(method),
            base64encode(method.I.toBinary(input, options.binaryOptions)),
            toNativeMetadata(requestMeta),
            deadlineMs(options.timeout),
        );

        return new ServerStreamingCall<I, O>(
            method,
            requestMeta,
            input,
            headers.promise,
            responses,
            status.promise,
            trailers.promise,
        );
    }

    clientStreaming<I extends object, O extends object>(method: MethodInfo<I, O>): ClientStreamingCall<I, O> {
        throw new RpcError(`Client streaming is not supported (${methodPath(method)}).`, "UNIMPLEMENTED");
    }

    duplex<I extends object, O extends object>(method: MethodInfo<I, O>): DuplexStreamingCall<I, O> {
        throw new RpcError(`Duplex streaming is not supported (${methodPath(method)}).`, "UNIMPLEMENTED");
    }
}
```

```ts
import { GrpcWebFetchTransport } from "@protobuf-ts/grpcweb-transport";
import { MethodInfo, RpcInterceptor, RpcOptions, RpcTransport } from "@protobuf-ts/runtime-rpc";
import { fetch as expoFetch } from "expo/fetch";
import { TransportMode } from "@/config/transport";
import { isAnonymousService } from "./status";
import { NativeGrpcTransport } from "./native/NativeGrpcTransport";
import { NativeGrpcBridge } from "./native/nativeGrpcBridge";

export interface GrpcTransportDeps {
    getBaseUrl: () => URL;
    getAccessToken: () => string | undefined;
    bridge: NativeGrpcBridge;
}

export const withAuthorization = (
    method: MethodInfo,
    options: RpcOptions,
    token: string | undefined,
): RpcOptions => {
    if (!token || isAnonymousService(method.service.typeName) || options.meta?.authorization) {
        return options;
    }

    return { ...options, meta: { ...options.meta, authorization: `Bearer ${token}` } };
};

export const createAuthInterceptor = (getAccessToken: () => string | undefined): RpcInterceptor => ({
    interceptUnary: (next, method, input, options) =>
        next(method, input, withAuthorization(method, options, getAccessToken())),
    interceptServerStreaming: (next, method, input, options) =>
        next(method, input, withAuthorization(method, options, getAccessToken())),
});

export const createGrpcTransport = (
    mode: Exclude<TransportMode, "Rest">,
    deps: GrpcTransportDeps,
): RpcTransport => {
    const interceptors = [createAuthInterceptor(deps.getAccessToken)];

    if (mode === "GrpcWeb") {
        // expo/fetch streams response bodies on Android; React Native's fetch buffers them,
        // which would break LiveUpdatesService.Subscribe.
        return new GrpcWebFetchTransport({
            baseUrl: deps.getBaseUrl().origin,
            format: "binary",
            fetch: expoFetch as unknown as typeof fetch,
            interceptors,
        });
    }

    return new NativeGrpcTransport({
        bridge: deps.bridge,
        getBaseUrl: () => deps.getBaseUrl().origin,
        interceptors,
    });
};
```

```ts
import { RpcError, RpcOptions, UnaryCall } from "@protobuf-ts/runtime-rpc";
import { ApiError, connectionApiError } from "@/api/apiError";
import { apiErrorFromRpcError } from "./grpcErrors";
import { classifyGrpcFailure, GrpcFailureKind } from "./status";

export const defaultGrpcTimeoutMs = 15000;

export type GrpcInvoke<O extends object> = (options: RpcOptions) => UnaryCall<any, O>;

export type GrpcOutcome<O> =
    | { isSuccess: true; response: O }
    | { isSuccess: false; kind: GrpcFailureKind; code: string; error: ApiError };

export interface GrpcCallDeps {
    isAuthenticated(): boolean;
    refresh(): Promise<string | null>;
    logout(): void;
    onSuccess(): void;
    onRefreshFailed(): void;
    onConsentMissing(error: ApiError): void;
}

const settle = async <O extends object>(
    invoke: GrpcInvoke<O>,
    options: RpcOptions,
): Promise<GrpcOutcome<O>> => {
    let call: UnaryCall<any, O> | undefined;

    try {
        call = invoke(options);
        const finished = await call;
        return { isSuccess: true, response: finished.response };
    } catch (e) {
        if (e instanceof RpcError) {
            return {
                isSuccess: false,
                kind: classifyGrpcFailure({
                    code: e.code,
                    // invoke itself may throw (for example UNIMPLEMENTED), leaving call undefined.
                    serviceTypeName: call?.method.service.typeName ?? e.serviceName ?? "",
                    methodName: call?.method.name ?? e.methodName ?? "",
                    detail: e.message,
                }),
                code: e.code,
                error: apiErrorFromRpcError(e),
            };
        }

        console.error("[gRPC] Call failed without a gRPC status", e);
        return { isSuccess: false, kind: "connectionError", code: "UNKNOWN", error: connectionApiError };
    }
};

/** Runs a unary call with the same session behavior HttpClientProvider applies to REST calls. */
export const executeGrpcCall = async <O extends object>(
    deps: GrpcCallDeps,
    invoke: GrpcInvoke<O>,
): Promise<GrpcOutcome<O>> => {
    const finish = (outcome: GrpcOutcome<O>): GrpcOutcome<O> => {
        if (outcome.isSuccess) {
            deps.onSuccess();
        } else if (outcome.kind === "consentMissing") {
            // LegalConsentInterceptor refusal: same as REST LegalMiddleware 403.
            deps.logout();
            deps.onConsentMissing(outcome.error);
        }
        return outcome;
    };

    const first = await settle(invoke, { timeout: defaultGrpcTimeoutMs });

    if (first.isSuccess || first.kind !== "refreshAndRetry") {
        return finish(first);
    }

    if (!deps.isAuthenticated()) {
        deps.logout();
        return first;
    }

    const token = await deps.refresh();

    if (!token) {
        deps.onRefreshFailed();
        return first;
    }

    // The session state behind the auth interceptor updates on the next render, so pass the new token explicitly.
    const second = await settle(invoke, {
        timeout: defaultGrpcTimeoutMs,
        meta: { authorization: `Bearer ${token}` },
    });

    return finish(second);
};
```
